      font-family: 'Segoe UI', sans-serif;
      position: relative;
    }
    /* Pantalla de estrellas */
    .estrellas-container {
      position: fixed;
      top: 0;
      left: 0;
      width: 100%;
      height: 100%;
      background: linear-gradient(to bottom, #000033, #000000);
      z-index: 10;
      opacity: 0;
      pointer-events: none;
      transition: opacity 1s ease-in-out;
    }
    .estrellas-container.active {
      opacity: 1;
      pointer-events: all;
    }
    .estrella {
      position: absolute;
      background: white;
      border-radius: 50%;
      animation: brillar 2s infinite ease-in-out;
    }
    @keyframes brillar {
      0%, 100% { opacity: 0.3; transform: scale(1); }
      50% { opacity: 1; transform: scale(1.2); }
    }
    /* Pantalla de lluvia */
    .lluvia-container {
      position: fixed;
      top: 0;
      left: 0;
      width: 100%;
      height: 100%;
      background: linear-gradient(to top right, #ff9a9e, #fad0c4);
      z-index: 5;
      opacity: 0;
      pointer-events: none;
      transition: opacity 1s ease-in-out;
    }
    .lluvia-container.active {
      opacity: 1;
      pointer-events: all;
    }
    h1 {
      color: white;
      text-shadow: 2px 2px 5px rgba(0,0,0,0.3);
      font-size: 2.5em;
      text-align: center;
      z-index: 6;
      position: relative;
      opacity: 0;
      transition: opacity 1s ease-in-out;
    }
    h1.visible {
      opacity: 1;
    }
    .heart {
      position: absolute;
      color: #ff3366;
      font-size: 24px;
      animation: float 6s linear infinite;
      pointer-events: none;
    }
    @keyframes float {
      0% { transform: translateY(100vh) scale(1); opacity: 1; }
      100% { transform: translateY(-10vh) scale(1.5); opacity: 0; }
    }
    /* Explosiones */
    .explosion {
      position: absolute;
      pointer-events: none;
      z-index: 1000;
    }
    .explosion-particle {
      position: absolute;
      width: 10px;
      height: 10px;
      background: radial-gradient(circle, #ffff00, #ff6b00);
      border-radius: 50%;
      animation: explode 0.8s ease-out forwards;
    }
    @keyframes explode {
      0% {
        transform: translate(0, 0) scale(1);
        opacity: 1;
      }
      100% {
        transform: translate(var(--tx), var(--ty)) scale(0);
        opacity: 0;
      }
    }
    /* Pantalla final negra */
    .pantalla-negra {
      position: fixed;

<body>
  <!-- Pantalla de estrellas -->
  <div class="estrellas-container" id="pantallaEstrellas"></div>
  <!-- Pantalla de lluvia -->
  <div class="lluvia-container" id="pantallaLluvia">
    <h1 id="mensaje">💖 TE AMO 💖</h1>
  </div>
  <!-- Pantalla final negra -->
  <div class="pantalla-negra" id="pantallaNegra"></div>
  <script>
    let enPantallaEstrellas = false;
    let enPantallaLluvia = false;
    let explosionesHabilitadas = false;
    // Crear estrellas
    function crearEstrellas() {
      const container = document.getElementById('pantallaEstrellas');
      for (let i = 0; i < 100; i++) {
        const estrella = document.createElement('div');
        estrella.className = 'estrella';
        estrella.style.left = Math.random() * 100 + '%';
        estrella.style.top = Math.random() * 100 + '%';
        estrella.style.width = (Math.random() * 3 + 1) + 'px';
        estrella.style.height = estrella.style.width;
        estrella.style.animationDelay = Math.random() * 2 + 's';
        container.appendChild(estrella);
      }
    }
    // Crear corazones flotantes
    function createHeart() {
      if (!enPantallaLluvia) return;
      const heart = document.createElement("div");
      heart.className = "heart";
      heart.innerHTML = "❤";
      heart.style.left = Math.random() * 100 + "vw";
      heart.style.animationDuration = (3 + Math.random() * 3) + "s";
      document.body.appendChild(heart);
      setTimeout(() => heart.remove(), 6000);
    }
    // Crear explosión
    function crearExplosion(x, y) {
      if (!explosionesHabilitadas) return;
      const explosion = document.createElement('div');
      explosion.className = 'explosion';
      explosion.style.left = x + 'px';
      explosion.style.top = y + 'px';
      // Crear partículas
      for (let i = 0; i < 20; i++) {
        const particle = document.createElement('div');
        particle.className = 'explosion-particle';
        const angle = (Math.PI * 2 * i) / 20;
        const distance = 50 + Math.random() * 50;
        const tx = Math.cos(angle) * distance;
        const ty = Math.sin(angle) * distance;
        particle.style.setProperty('--tx', tx + 'px');
        particle.style.setProperty('--ty', ty + 'px');
        explosion.appendChild(particle);
      }
      document.body.appendChild(explosion);
      setTimeout(() => explosion.remove(), 1000);
    }
    // Habilitar explosiones
    function habilitarExplosiones() {
      if (enPantallaLluvia) {
        explosionesHabilitadas = true;
      }
    }
    // Deshabilitar explosiones
    function deshabilitarExplosiones() {
      explosionesHabilitadas = false;
    }
    // Event listener para clicks
    document.addEventListener('click', function(e) {
      if (explosionesHabilitadas && enPantallaLluvia) {
        crearExplosion(e.clientX, e.clientY);
      }
    });
    // Secuencia de animación
    function iniciarAnimacion() {
      // Crear estrellas primero
      crearEstrellas();
      // Mostrar pantalla de estrellas
      setTimeout(() => {
        enPantallaEstrellas = true;
        document.getElementById('pantallaEstrellas').classList.add('active');
      }, 500);
      // Cambiar a pantalla de lluvia después de 30 segundos (30000ms)
      setTimeout(() => {
        // Ocultar estrellas
        enPantallaEstrellas = false;
        document.getElementById('pantallaEstrellas').classList.remove('active');
        // Esperar transición (1.5s de la transición de estrellas)
        setTimeout(() => {
          // Mostrar lluvia
          enPantallaLluvia = true;
          document.getElementById('pantallaLluvia').classList.add('active');
          document.getElementById('mensaje').classList.add('visible');
          // Habilitar explosiones solo en pantalla de lluvia
          habilitarExplosiones();
          // Iniciar corazones
          const heartInterval = setInterval(createHeart, 500);
          // Después de 1.5s más, ir a pantalla negra (total 3000ms de transiciones)
          setTimeout(() => {
            // Deshabilitar explosiones
            deshabilitarExplosiones();
            // Detener corazones
            clearInterval(heartInterval);
            // Ocultar lluvia
            enPantallaLluvia = false;
            document.getElementById('pantallaLluvia').classList.remove('active');
            document.getElementById('mensaje').classList.remove('visible');
            // Mostrar pantalla negra
            setTimeout(() => {
              document.getElementById('pantallaNegra').classList.add('active');
            }, 1000);
          }, 1500);
        }, 1500);
      }, 30000); // 30 segundos para estrellas
    }
    // Iniciar cuando la página carga
    window.addEventListener('load', iniciarAnimacion);
  </script>
